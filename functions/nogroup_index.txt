// ============================================================================
// V2 / NO-GROUP TRIGGERS
// ESKI TRIGGERLAR YUKARIDA AYNEN KALIYOR
// ============================================================================

async function getCountryCodeForRootTarget(
    collectionName,
    targetId,
) {
  try {
    const targetSnap = await admin
        .firestore()
        .collection(collectionName)
        .doc(targetId)
        .get();

    return targetSnap.data()?.countryCode || "US";
  } catch (error) {
    console.error(
        "ROOT COUNTRY CODE ERROR",
        error,
    );

    return "US";
  }
}


// ============================================================================
// V2 CALL ME
// REQUESTER -> LOCATOR
//
// locators/{locatorId}/call_me/items/{callMeId}
// ============================================================================

exports.onLocatorCallMeCreatedV2 = onDocumentCreated(
    "locators/{locatorId}/call_me/items/{callMeId}",

    async (event) => {
      const data = event.data.data();

      const locatorId =
          event.params.locatorId;

      const callMeId =
          event.params.callMeId;

      const createdAtMillis =
          data.createdAt?.toMillis?.() ??
          Date.now();

      const requesterName =
          data.requesterName ||
          "Requester";

      const requesterCode =
          data.requesterCode ||
          "";

      const topic =
          `locator_${locatorId}`;

      let text = messages.en;

      try {
        const countryCode =
            await getCountryCodeForRootTarget(
                "locators",
                locatorId,
            );

        const lang =
            getLanguageFromCountry(
                countryCode,
            );

        text =
            messages[lang] ||
            messages.en;
      } catch (error) {
        console.error(
            "CALL ME V2 REQUESTER->LOCATOR LOCALIZATION ERROR",
            error,
        );
      }

      const notificationTitle =
          text.callMeTitle;

      const notificationBody =
          text.callMeBody(
              requesterName,
          );

      try {
        const response =
            await admin.messaging().send({
              topic,

              android: {
                priority: "high",
              },

              data: {
                type: "call_me",

                callMeId,

                createdAt:
                    createdAtMillis
                        .toString(),

                requesterName,
                requesterCode,

                title:
                    notificationTitle,

                body:
                    notificationBody,
              },
            });

        console.log(
            "CALL ME V2 REQUESTER->LOCATOR SENT",
            topic,
            response,
        );
      } catch (error) {
        console.error(
            "CALL ME V2 REQUESTER->LOCATOR ERROR",
            error,
        );
      }
    },
);

// ============================================================================
// V2 ACTIVE WATCHERS
//
// presence/active_watchers/{locatorId}/{requesterId}
// ============================================================================

exports.onActiveWatchersChangedV2 = onValueWritten(
    "/presence/active_watchers/{locatorId}",

    async (event) => {
      const locatorId =
          event.params.locatorId;

      const before =
          event.data.before.val() ||
          {};

      const after =
          event.data.after.val() ||
          {};

      const beforeCount =
          Object.keys(before).length;

      const afterCount =
          Object.keys(after).length;

      console.log(
          "ACTIVE WATCHERS V2 CHANGED",
          "locatorId=",
          locatorId,
          "beforeCount=",
          beforeCount,
          "afterCount=",
          afterCount,
      );

      if (beforeCount === afterCount) {
        console.log(
            "ACTIVE WATCHERS V2 => count unchanged, skip FCM",
        );

        return;
      }

      const topic =
          `locator_${locatorId}`;

      try {
        const response =
            await admin.messaging().send({
              topic,

              android: {
                priority: "high",
              },

              data: {
                type:
                    "active_watchers_changed",
              },
            });

        console.log(
            "ACTIVE WATCHERS V2 FCM SENT",
            topic,
            response,
        );
      } catch (error) {
        console.error(
            "ACTIVE WATCHERS V2 FCM ERROR",
            error,
        );
      }
    },
);
